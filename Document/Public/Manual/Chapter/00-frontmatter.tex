{\rmfamily\color{rvmmid}Technical Manual}\\[12pt]
\end{center}

\begin{description}[leftmargin=0pt,labelindent=0pt,style=nextline,itemsep=8pt]
    \item[\textsf{\color{rvmpurple}High-Performance Real-Time Response}]
    Native processes provide excellent real-time performance.
    Virtual machine processes provide excellent compatibility and execution efficiency.

    \item[\textsf{\color{rvmpurple}Broad Architecture Support}]
    Easily portable across multiple architectures.
    Supports paravirtualization on microcontrollers.

    \item[\textsf{\color{rvmpurple}High Security}]
    Hardware isolation between virtual machines.
    Supports fault restart.

    \item[\textsf{\color{rvmpurple}Comprehensive Ecosystem}]
    Supports virtualization of most operating systems.
    Original system applications run without modification.

    \item[\textsf{\color{rvmpurple}High Usability}]
    Provides a graphical project generator.
\end{description}
